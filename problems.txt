// src/lib/axios/config.js (YENİ DOSYA)

import axios from 'axios';

const API_BASE_URL = process.env.NEXT_PUBLIC_API_URL;

// E-ticaret için farklı timeout stratejileri
const TIMEOUT_CONFIG = {
  DEFAULT: 15000,      // Genel işlemler
  PAYMENT: 30000,      // Ödeme işlemleri (3D Secure)
  UPLOAD: 60000,       // Dosya yükleme
  SEARCH: 5000,        // Arama istekleri
};

// Retry stratejisi
const RETRY_CONFIG = {
  retries: 3,
  retryDelay: (retryCount) => {
    return Math.min(1000 * 2 ** retryCount, 10000); // Exponential backoff
  },
  retryCondition: (error) => {
    // Sadece network hatalarında retry
    return !error.response && error.code !== 'ECONNABORTED';
  },
};

// Ana axios instance
export const instance = axios.create({
  baseURL: API_BASE_URL,
  timeout: TIMEOUT_CONFIG.DEFAULT,
  headers: {
    'Content-Type': 'application/json',
  },
});

// Payment-specific instance
export const paymentInstance = axios.create({
  baseURL: API_BASE_URL,
  timeout: TIMEOUT_CONFIG.PAYMENT,
  headers: {
    'Content-Type': 'application/json',
  },
});

// Upload-specific instance
export const uploadInstance = axios.create({
  baseURL: API_BASE_URL,
  timeout: TIMEOUT_CONFIG.UPLOAD,
});

// Request interceptor - JWT token
instance.interceptors.request.use(
  (config) => {
    const token = localStorage.getItem('token');
    if (token) {
      config.headers.Authorization = `Bearer ${token}`;
    }
    return config;
  },
  (error) => Promise.reject(error)
);

// Response interceptor - Error handling
instance.interceptors.response.use(
  (response) => response,
  async (error) => {
    const originalRequest = error.config;

    // Token expired - refresh
    if (error.response?.status === 401 && !originalRequest._retry) {
      originalRequest._retry = true;
      try {
        const refreshToken = localStorage.getItem('refreshToken');
        const response = await axios.post(`${API_BASE_URL}/auth/refresh`, {
          refreshToken,
        });
        localStorage.setItem('token', response.data.accessToken);
        originalRequest.headers.Authorization = `Bearer ${response.data.accessToken}`;
        return instance(originalRequest);
      } catch (refreshError) {
        // Refresh failed - logout user
        localStorage.clear();
        window.location.href = '/login';
        return Promise.reject(refreshError);
      }
    }

    // Retry logic for network errors
    if (
      !originalRequest._retry &&
      RETRY_CONFIG.retryCondition(error) &&
      (originalRequest._retryCount || 0) < RETRY_CONFIG.retries
    ) {
      originalRequest._retryCount = (originalRequest._retryCount || 0) + 1;
      const delay = RETRY_CONFIG.retryDelay(originalRequest._retryCount);
      
      await new Promise((resolve) => setTimeout(resolve, delay));
      return instance(originalRequest);
    }

    return Promise.reject(error);
  }
);

// Tüm instance'lara aynı interceptor'ları ekle
[paymentInstance, uploadInstance].forEach(inst => {
  inst.interceptors.request.use(instance.interceptors.request.handlers[0].fulfilled);
  inst.interceptors.response.use(
    instance.interceptors.response.handlers[0].fulfilled,
    instance.interceptors.response.handlers[0].rejected
  );
});

'use client';

import React from 'react';
import { AlertTriangle, RefreshCw, Home } from 'lucide-react';
import { Button } from '@/components/ui/button';

class ErrorBoundary extends React.Component {
  constructor(props) {
    super(props);
    this.state = { 
      hasError: false, 
      error: null,
      errorInfo: null,
      errorCount: 0
    };
  }

  static getDerivedStateFromError(error) {
    return { hasError: true };
  }

  componentDidCatch(error, errorInfo) {
    // Log to error reporting service (Sentry, LogRocket)
    console.error('ErrorBoundary caught:', error, errorInfo);
    
    this.setState(prevState => ({
      error,
      errorInfo,
      errorCount: prevState.errorCount + 1
    }));

    // E-ticaret için özel: Sepet backup
    if (typeof window !== 'undefined') {
      const cart = localStorage.getItem('cart');
      if (cart) {
        localStorage.setItem('cart_backup', cart);
        console.log('🛡️ Cart backed up before crash');
      }
    }

    // Production'da error reporting
    if (process.env.NODE_ENV === 'production') {
      // Sentry.captureException(error);
    }
  }

  handleReset = () => {
    // Sepet restore
    if (typeof window !== 'undefined') {
      const cartBackup = localStorage.getItem('cart_backup');
      if (cartBackup) {
        localStorage.setItem('cart', cartBackup);
        console.log('🔄 Cart restored from backup');
      }
    }

    this.setState({ 
      hasError: false, 
      error: null, 
      errorInfo: null 
    });
  };

  handleGoHome = () => {
    window.location.href = '/';
  };

  render() {
    if (this.state.hasError) {
      // Too many errors - show minimal UI
      if (this.state.errorCount > 3) {
        return (
          <div className="min-h-screen flex items-center justify-center bg-gray-50 px-4">
            <div className="text-center">
              <AlertTriangle className="w-16 h-16 text-red-500 mx-auto mb-4" />
              <h1 className="text-2xl font-bold mb-2">Ciddi Bir Hata Oluştu</h1>
              <p className="text-gray-600 mb-6">
                Lütfen sayfayı yenileyerek tekrar deneyin.
              </p>
              <Button onClick={() => window.location.reload()}>
                Sayfayı Yenile
              </Button>
            </div>
          </div>
        );
      }

      // E-ticaret için özel error UI
      const isCheckout = window.location.pathname.includes('checkout');
      const isPayment = window.location.pathname.includes('payment');

      return (
        <div className="min-h-screen flex items-center justify-center bg-gray-50 px-4">
          <div className="max-w-md w-full bg-white rounded-lg shadow-lg p-8">
            <div className="text-center">
              <AlertTriangle className="w-16 h-16 text-yellow-500 mx-auto mb-4" />
              <h1 className="text-2xl font-bold mb-2">Bir Şeyler Ters Gitti</h1>
              
              {isCheckout || isPayment ? (
                <div className="bg-blue-50 border border-blue-200 rounded-lg p-4 mb-6">
                  <p className="text-sm text-blue-800">
                    ✅ Sepetiniz güvende! Verileriniz kaybedilmedi.
                  </p>
                </div>
              ) : null}

              <p className="text-gray-600 mb-6">
                {isCheckout 
                  ? "Ödeme işleminiz sırasında bir hata oluştu. Sepetiniz kayıtlı, lütfen tekrar deneyin."
                  : "Beklenmeyen bir hata oluştu. Lütfen sayfayı yenileyin."}
              </p>

              {process.env.NODE_ENV === 'development' && (
                <details className="text-left mb-6 bg-gray-100 p-4 rounded text-xs">
                  <summary className="cursor-pointer font-semibold mb-2">
                    Hata Detayları (Development)
                  </summary>
                  <pre className="overflow-auto">
                    {this.state.error?.toString()}
                    {'\n\n'}
                    {this.state.errorInfo?.componentStack}
                  </pre>
                </details>
              )}

              <div className="flex gap-3">
                <Button 
                  onClick={this.handleReset}
                  className="flex-1"
                  variant="default"
                >
                  <RefreshCw className="w-4 h-4 mr-2" />
                  Tekrar Dene
                </Button>
                <Button 
                  onClick={this.handleGoHome}
                  className="flex-1"
                  variant="outline"
                >
                  <Home className="w-4 h-4 mr-2" />
                  Ana Sayfa
                </Button>
              </div>
            </div>
          </div>
        </div>
      );
    }

    return this.props.children;
  }
}

export default ErrorBoundary;

// src/lib/utils/cartPersistence.js (YENİ DOSYA)

const CART_STORAGE_KEY = 'cart';
const CART_BACKUP_KEY = 'cart_backup';
const CART_SYNC_KEY = 'cart_sync';
const CART_EXPIRY_DAYS = 7;

// Cart persistence utilities
export const cartStorage = {
  // Save cart to localStorage
  save: (cart) => {
    try {
      const cartData = {
        items: cart,
        timestamp: Date.now(),
        expiresAt: Date.now() + (CART_EXPIRY_DAYS * 24 * 60 * 60 * 1000)
      };
      localStorage.setItem(CART_STORAGE_KEY, JSON.stringify(cartData));
      
      // Trigger sync event for other tabs
      localStorage.setItem(CART_SYNC_KEY, Date.now().toString());
      
      return true;
    } catch (error) {
      console.error('Cart save error:', error);
      return false;
    }
  },

  // Load cart from localStorage
  load: () => {
    try {
      const cartDataStr = localStorage.getItem(CART_STORAGE_KEY);
      if (!cartDataStr) return [];

      const cartData = JSON.parse(cartDataStr);
      
      // Check expiry
      if (cartData.expiresAt && Date.now() > cartData.expiresAt) {
        cartStorage.clear();
        return [];
      }

      return cartData.items || [];
    } catch (error) {
      console.error('Cart load error:', error);
      return [];
    }
  },

  // Backup cart (before risky operations)
  backup: () => {
    try {
      const currentCart = localStorage.getItem(CART_STORAGE_KEY);
      if (currentCart) {
        localStorage.setItem(CART_BACKUP_KEY, currentCart);
        console.log('🛡️ Cart backed up');
      }
    } catch (error) {
      console.error('Cart backup error:', error);
    }
  },

  // Restore cart from backup
  restore: () => {
    try {
      const backupCart = localStorage.getItem(CART_BACKUP_KEY);
      if (backupCart) {
        localStorage.setItem(CART_STORAGE_KEY, backupCart);
        console.log('🔄 Cart restored from backup');
        return JSON.parse(backupCart).items || [];
      }
      return [];
    } catch (error) {
      console.error('Cart restore error:', error);
      return [];
    }
  },

  // Clear cart
  clear: () => {
    try {
      localStorage.removeItem(CART_STORAGE_KEY);
      localStorage.removeItem(CART_BACKUP_KEY);
    } catch (error) {
      console.error('Cart clear error:', error);
    }
  },

  // Merge carts (for multi-tab sync)
  merge: (localCart, syncedCart) => {
    const mergedMap = new Map();

    // Add local cart items
    localCart.forEach(item => {
      mergedMap.set(item.id, item);
    });

    // Merge synced cart items
    syncedCart.forEach(item => {
      if (mergedMap.has(item.id)) {
        // Take the one with higher count
        const existing = mergedMap.get(item.id);
        if (item.count > existing.count) {
          mergedMap.set(item.id, item);
        }
      } else {
        mergedMap.set(item.id, item);
      }
    });

    return Array.from(mergedMap.values());
  }
};

// src/lib/utils/paymentRecovery.js (YENİ DOSYA)

const PAYMENT_STATE_KEY = 'payment_state';

export const paymentRecovery = {
  // Save payment state before 3D Secure redirect
  saveState: (orderId, paymentId, amount) => {
    const state = {
      orderId,
      paymentId,
      amount,
      timestamp: Date.now(),
      status: 'PENDING'
    };
    localStorage.setItem(PAYMENT_STATE_KEY, JSON.stringify(state));
  },

  // Load payment state after redirect
  loadState: () => {
    try {
      const stateStr = localStorage.getItem(PAYMENT_STATE_KEY);
      if (!stateStr) return null;
      return JSON.parse(stateStr);
    } catch (error) {
      return null;
    }
  },

  // Clear payment state
  clearState: () => {
    localStorage.removeItem(PAYMENT_STATE_KEY);
  },

  // Check for incomplete payment on app load
  checkIncompletePayment: async (api) => {
    const state = paymentRecovery.loadState();
    if (!state) return null;

    // Payment state exists - check if it was completed
    const timeDiff = Date.now() - state.timestamp;
    const TIMEOUT = 10 * 60 * 1000; // 10 dakika

    if (timeDiff > TIMEOUT) {
      // Too old - clear it
      paymentRecovery.clearState();
      return null;
    }

    try {
      // Check payment status from backend
      const response = await api.get(`/payment/${state.paymentId}/status`);
      
      if (response.data.status === 'SUCCESS') {
        // Payment completed - redirect to success
        return {
          type: 'SUCCESS',
          orderId: state.orderId,
          paymentId: state.paymentId
        };
      } else if (response.data.status === 'FAILED') {
        // Payment failed - show error
        paymentRecovery.clearState();
        return {
          type: 'FAILED',
          error: response.data.errorMessage
        };
      } else {
        // Still pending - user can continue waiting
        return {
          type: 'PENDING',
          paymentId: state.paymentId
        };
      }
    } catch (error) {
      console.error('Payment recovery check error:', error);
      return null;
    }
  }
};

const submitOrder = async (formData) => {

    } else {
      // Cash/Gift Card - direct success

      clearCart();
      router.push(`/payment/success?orderId=${orderResponse.id}`);

    // Payment error - restore cart
    const restoredCart = cartStorage.restore();
    if (restoredCart.length > 0) {
      dispatch(loadCartFromStorageAction(restoredCart));
      toast.success("Sepetiniz geri yüklendi");
    }
    
    toast.error(error.message || "Ödeme işlemi başarısız");
  }
};

// src/lib/utils/notifications.js (YENİ)

export const notificationManager = {
  permission: null,

  async requestPermission() {
    if (!('Notification' in window)) {
      console.warn('Tarayıcı bildirim desteklemiyor');
      return false;
    }

    if (Notification.permission === 'granted') {
      this.permission = 'granted';
      return true;
    }

    if (Notification.permission !== 'denied') {
      const permission = await Notification.requestPermission();
      this.permission = permission;
      return permission === 'granted';
    }

    return false;
  },

  show(title, options = {}) {
    if (this.permission !== 'granted') return;

    const notification = new Notification(title, {
      icon: '/logo.png',
      badge: '/badge.png',
      ...options
    });

    notification.onclick = () => {
      window.focus();
      notification.close();
    };

    return notification;
  },

  showNewOrder(order) {
    this.show('Yeni Sipariş!', {
      body: `Sipariş #${order.id} - ${order.totalAmount} TL`,
      tag: `order-${order.id}`,
      requireInteraction: true
    });
  }
};

// src/lib/hooks/useLoadingManager.js (YENİ)

import { useDispatch, useSelector } from 'react-redux';
import { setModuleLoading } from '@/lib/store/actions/globalActions';

export const useLoadingManager = (module) => {
  const dispatch = useDispatch();
  const loadingStates = useSelector(state => state.global.moduleLoading);
  
  const setLoading = (loading) => {
    dispatch(setModuleLoading({ module, loading }));
  };

  const isLoading = loadingStates[module] || false;
  const isAnyLoading = Object.values(loadingStates).some(l => l);

  return { 
    isLoading, 
    isAnyLoading,
    setLoading 
  };
};

// src/lib/analytics/tracker.js (YENİ)

class AnalyticsTracker {
  constructor() {
    this.enabled = process.env.NODE_ENV === 'production';
  }

  // E-commerce events
  trackProductView(product) {
    if (!this.enabled) return;
    
    // Google Analytics 4
    if (typeof window !== 'undefined' && window.gtag) {
      window.gtag('event', 'view_item', {
        currency: 'TRY',
        value: product.price,
        items: [{
          item_id: product.id,
          item_name: product.name,
          price: product.price,
          item_category: product.categoryName
        }]
      });
    }
  }

  trackAddToCart(product, quantity) {
    if (!this.enabled) return;
    
    if (typeof window !== 'undefined' && window.gtag) {
      window.gtag('event', 'add_to_cart', {
        currency: 'TRY',
        value: product.price * quantity,
        items: [{
          item_id: product.id,
          item_name: product.name,
          price: product.price,
          quantity: quantity
        }]
      });
    }
  }

  trackPurchase(order) {
    if (!this.enabled) return;
    
    if (typeof window !== 'undefined' && window.gtag) {
      window.gtag('event', 'purchase', {
        transaction_id: order.id,
        value: order.totalAmount,
        currency: 'TRY',
        items: order.items.map(item => ({
          item_id: item.productId,
          item_name: item.productName,
          price: item.price,
          quantity: item.quantity
        }))
      });
    }
  }

  trackCheckoutStep(step, paymentMethod = null) {
    if (!this.enabled) return;
    
    const eventData = {
      checkout_step: step,
      checkout_option: step === 3 ? paymentMethod : null
    };
    
    if (typeof window !== 'undefined' && window.gtag) {
      window.gtag('event', 'checkout_progress', eventData);
    }
  }
}

export const analytics = new AnalyticsTracker();
