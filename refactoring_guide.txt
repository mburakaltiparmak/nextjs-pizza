NEXT.JS PIZZA FRONTEND REFACTORING GUIDE FOR AI AGENT
================================================================================

PROJECT CONTEXT
---------------
- Current State: Working Next.js 15 App Router application with Redux state management
- Backend: Spring Boot REST API (running locally, no changes allowed)
- Goal: Restructure frontend to follow Next.js best practices based on switch-high-tech reference project
- Constraint: No backend modifications, maintain all existing functionality, stay with JavaScript (no TypeScript migration)
- Redux Pattern: Action-Reducer pattern (NOT Redux Toolkit slices)
- Component Pattern: Server/Client component separation with page-based organization

REFERENCE ARCHITECTURE ANALYSIS (switch-high-tech)
================================================================================

DIRECTORY STRUCTURE PRINCIPLES
-------------------------------
src/app/
├── (admin)/          # Route group for admin pages
├── (public)/         # Route group for public pages
├── _components/      # Shared components (not routable)
├── _lib/            # Utilities, hooks (not routable)
└── api/             # API routes

KEY PATTERNS OBSERVED
---------------------
1. Route Groups: Use parentheses for logical grouping without affecting URL structure
2. Private Folders: Prefix with underscore (_) for non-routable shared code
3. Server/Client Separation: Clear distinction between server and client components
4. Colocation: Keep page-specific components near their pages
5. Component Organization: Page-specific components in dedicated folders

================================================================================
SERVER VS CLIENT COMPONENT STRATEGY
================================================================================

SERVER COMPONENTS (DEFAULT)
----------------------------
All page.jsx files are SERVER components by default

Example - app/(public)/menu/page.jsx (SERVER COMPONENT):
---------------------------------------------------------
// NO "use client" directive

export default async function MenuPage() {
  // Can fetch data on server if needed (optional)
  // const categories = await fetchCategoriesFromAPI();
  
  // Pass to client component
  return <MenuClient />;
}

Benefits of Server Components:
- Can fetch data on server (optional, we mostly use Redux)
- Reduced JavaScript bundle size
- Better SEO
- Faster initial page load

CLIENT COMPONENTS
-----------------
All interactive components need "use client" directive

Example - app/(public)/menu/MenuClient.jsx (CLIENT COMPONENT):
---------------------------------------------------------------
"use client";

import { useEffect } from 'react';
import { useDispatch, useSelector } from 'react-redux';
import { fetchProducts } from '@/lib/store/actions/productActions';

export default function MenuClient() {
  const dispatch = useDispatch();
  const products = useSelector(state => state.products.products);
  
  useEffect(() => {
    dispatch(fetchProducts());
  }, [dispatch]);
  
  return (
    <div>
      {/* Interactive UI */}
    </div>
  );
}

When to use Client Components:
- Need React hooks (useState, useEffect, useReducer)
- Need Redux (useSelector, useDispatch)
- Need event handlers (onClick, onChange)
- Need browser APIs (window, localStorage)
- Need interactivity

COMPONENT ORGANIZATION PATTERN
-------------------------------
app/(public)/menu/
├── page.jsx              # SERVER component (minimal, just returns MenuClient)
└── MenuClient.jsx        # CLIENT component (all logic, Redux, hooks)

components/menu/          # All sub-components (CLIENT)

// Action Types
export const FETCH_PRODUCTS_REQUEST = 'FETCH_PRODUCTS_REQUEST';
export const FETCH_PRODUCTS_SUCCESS = 'FETCH_PRODUCTS_SUCCESS';
export const FETCH_PRODUCTS_FAILURE = 'FETCH_PRODUCTS_FAILURE';

export const FETCH_PRODUCT_BY_ID_REQUEST = 'FETCH_PRODUCT_BY_ID_REQUEST';
export const FETCH_PRODUCT_BY_ID_SUCCESS = 'FETCH_PRODUCT_BY_ID_SUCCESS';
export const FETCH_PRODUCT_BY_ID_FAILURE = 'FETCH_PRODUCT_BY_ID_FAILURE';

export const SET_CURRENT_PRODUCT = 'SET_CURRENT_PRODUCT';
export const CLEAR_CURRENT_PRODUCT = 'CLEAR_CURRENT_PRODUCT';

// Action Creators (Thunks)
export const fetchProducts = () => async (dispatch) => {
  dispatch({ type: FETCH_PRODUCTS_REQUEST });
  
  try {
    const response = await api.get('/products');
    dispatch({ 
      type: FETCH_PRODUCTS_SUCCESS, 
      payload: response.data 
    });
  } catch (error) {
    dispatch({ 
      type: FETCH_PRODUCTS_FAILURE, 
      payload: error.response?.data?.message || 'Failed to fetch products' 
    });
  }
};

export const fetchProductById = (id) => async (dispatch) => {
  dispatch({ type: FETCH_PRODUCT_BY_ID_REQUEST });
  
  try {
    const response = await api.get(`/products/${id}`);
    dispatch({ 
      type: FETCH_PRODUCT_BY_ID_SUCCESS, 
      payload: response.data 
    });
    dispatch({ 
      type: SET_CURRENT_PRODUCT, 
      payload: response.data 
    });
  } catch (error) {
    dispatch({ 
      type: FETCH_PRODUCT_BY_ID_FAILURE, 
      payload: error.response?.data?.message || 'Failed to fetch product' 
    });
  }
};

export const setCurrentProduct = (product) => ({
  type: SET_CURRENT_PRODUCT,
  payload: product
});

export const clearCurrentProduct = () => ({
  type: CLEAR_CURRENT_PRODUCT

// lib/store/reducers/productReducer.js
import {
  FETCH_PRODUCTS_REQUEST,
  FETCH_PRODUCTS_SUCCESS,
  FETCH_PRODUCTS_FAILURE,
  FETCH_PRODUCT_BY_ID_REQUEST,
  FETCH_PRODUCT_BY_ID_SUCCESS,
  FETCH_PRODUCT_BY_ID_FAILURE,
  SET_CURRENT_PRODUCT,
  CLEAR_CURRENT_PRODUCT
} from '../actions/productActions';

const initialState = {
  products: [],
  currentProduct: null,  // New: for Problem #18
  loading: false,
  error: null
};

export default function productReducer(state = initialState, action) {
  switch (action.type) {
    case FETCH_PRODUCTS_REQUEST:
    case FETCH_PRODUCT_BY_ID_REQUEST:
      return {
        ...state,
        loading: true,
        error: null
      };
      
    case FETCH_PRODUCTS_SUCCESS:
      return {
        ...state,
        loading: false,
        products: action.payload,
        error: null
      };
      
    case FETCH_PRODUCT_BY_ID_SUCCESS:
      return {
        ...state,
        loading: false,
        error: null
      };
      
    case SET_CURRENT_PRODUCT:
      return {
        ...state,
        currentProduct: action.payload
      };
      
    case CLEAR_CURRENT_PRODUCT:
      return {
        ...state,
        currentProduct: null
      };
      
    case FETCH_PRODUCTS_FAILURE:
    case FETCH_PRODUCT_BY_ID_FAILURE:
      return {
        ...state,
        loading: false,
        error: action.payload
      };
      
    default:
      return state;
  }
}

// lib/store/reducers/globalReducer.js
const SET_GLOBAL_LOADING = 'SET_GLOBAL_LOADING';
const SET_GLOBAL_ERROR = 'SET_GLOBAL_ERROR';
const CLEAR_GLOBAL_ERROR = 'CLEAR_GLOBAL_ERROR';
const SET_INITIALIZED = 'SET_INITIALIZED';

const initialState = {
  globalLoading: false,  // App-wide loading overlay
  globalError: null,
  initialized: false
};

export default function globalReducer(state = initialState, action) {
  switch (action.type) {
    case SET_GLOBAL_LOADING:
      return {
        ...state,
        globalLoading: action.payload
      };
      
    case SET_GLOBAL_ERROR:
      return {
        ...state,
        globalError: action.payload
      };
      
    case CLEAR_GLOBAL_ERROR:
      return {
        ...state,
        globalError: null
      };
      
    case SET_INITIALIZED:
      return {
        ...state,
        initialized: action.payload
      };
      
    default:
      return state;
  }
}

export const setGlobalLoading = (loading) => ({
  type: SET_GLOBAL_LOADING,

export const setGlobalError = (error) => ({
  type: SET_GLOBAL_ERROR,
  payload: error
});

export const clearGlobalError = () => ({
  type: CLEAR_GLOBAL_ERROR
});

export const setInitialized = (initialized) => ({
  type: SET_INITIALIZED,
  payload: initialized

  (config) => {
    // Problem #14: Consistent FormData handling
    if (config.data instanceof FormData) {
      delete config.headers['Content-Type'];
    }

  (error) => Promise.reject(error)

  (response) => response,

const errorMiddleware = (store) => (next) => (action) => {
  // Check if action has error
  if (action.type.endsWith('_FAILURE') || action.error) {
    const errorMessage = action.payload || action.error?.message || 'An error occurred';
    
    console.error('Redux Error:', action.type, errorMessage);
    
    // Optionally dispatch to global error state
    store.dispatch({
      type: 'SET_GLOBAL_ERROR',
      payload: errorMessage
    });
    
    // You can add toast notification here if you have a toast library
    // toast.error(errorMessage);
  }
  
  return next(action);
};

import { useState, useEffect } from 'react';

export function useDebounce(value, delay = 300) {
  const [debouncedValue, setDebouncedValue] = useState(value);

  useEffect(() => {
    const handler = setTimeout(() => {
      setDebouncedValue(value);
    }, delay);

    return () => {
      clearTimeout(handler);
    };
  }, [value, delay]);

  return debouncedValue;
}

"use client";

import { useEffect, useRef } from 'react';
import { useDispatch, useSelector } from 'react-redux';
import { fetchDashboard } from '@/lib/store/actions/adminActions';

const MAX_RETRIES = 3;
const INITIAL_RETRY_DELAY = 2000; // Problem #2: Less aggressive backoff

export function useDashboardDataLoader() {
  const dispatch = useDispatch();
  const dataInitialized = useSelector(state => state.admin.dataInitialized);
  const retryCountRef = useRef(0);
  const isMountedRef = useRef(true);
  
  useEffect(() => {
    isMountedRef.current = true;
    
    const loadData = async () => {
      if (dataInitialized) return;
      
      try {
        // Problem #1, #19: Optimized parallel fetch with simple endpoint
        await dispatch(fetchDashboard());
        retryCountRef.current = 0;
      } catch (error) {
        if (retryCountRef.current < MAX_RETRIES && isMountedRef.current) {
          retryCountRef.current++;
          const delay = INITIAL_RETRY_DELAY * retryCountRef.current; // Linear backoff
          
          setTimeout(() => {
            if (isMountedRef.current) {
              loadData();
            }
          }, delay);
        }
      }
    };
    
    loadData();
    
    return () => {
      isMountedRef.current = false;
    };
  }, [dispatch, dataInitialized]);
}

"use client";

import { useEffect, useRef, useMemo, useState } from 'react';
import { useDispatch, useSelector } from 'react-redux';
import { fetchOrders, updateOrderStatus } from '@/lib/store/actions/orderActions';
import { useDebounce } from '@/lib/hooks/useDebounce';
import { OrderTable } from '@/components/orders-admin/OrderTable';
import { OrderFilters } from '@/components/orders-admin/OrderFilters';

const POLLING_INTERVAL = 30000; // Problem #6: Less aggressive polling (30s instead of 5s)

export default function OrdersAdminClient() {
  const dispatch = useDispatch();
  const orders = useSelector(state => state.orders.orders);
  const [searchTerm, setSearchTerm] = useState('');
  const [statusFilter, setStatusFilter] = useState('all');
  
  const mountedRef = useRef(true);
  const lastFetchRef = useRef(0);
  const pollingTimeoutRef = useRef(null);
  
  // Problem #7: Debounced search
  const debouncedSearch = useDebounce(searchTerm, 300);
  
  // Problem #7: Memoized filtered orders
  const filteredOrders = useMemo(() => {
    let result = orders;
    
    if (debouncedSearch) {
      result = result.filter(order => 
        order.id.toString().includes(debouncedSearch) ||
        order.customerName?.toLowerCase().includes(debouncedSearch.toLowerCase())
      );
    }
    
    if (statusFilter !== 'all') {
      result = result.filter(order => order.status === statusFilter);
    }
    
    return result;
  }, [orders, debouncedSearch, statusFilter]);
  
  // Problem #6: Smarter polling with cleanup
  useEffect(() => {
    mountedRef.current = true;
    
    const fetchData = async () => {
      const now = Date.now();
      if (now - lastFetchRef.current < 5000) return; // Minimum 5s between fetches
      
      lastFetchRef.current = now;
      await dispatch(fetchOrders());
      
      // Schedule next poll
      if (mountedRef.current) {
        pollingTimeoutRef.current = setTimeout(fetchData, POLLING_INTERVAL);
      }
    };
    
    fetchData();
    
    // Problem #8: Proper cleanup
    return () => {
      mountedRef.current = false;
      if (pollingTimeoutRef.current) {
        clearTimeout(pollingTimeoutRef.current);
        pollingTimeoutRef.current = null;
      }
    };
  }, [dispatch]);
  
  // Problem #9: Optimistic update
  const handleStatusUpdate = async (orderId, newStatus) => {
    // Optimistic update
    dispatch({
      type: 'UPDATE_ORDER_STATUS_OPTIMISTIC',
      payload: { orderId, status: newStatus }
    });
    
    try {
      await dispatch(updateOrderStatus(orderId, newStatus));
    } catch (error) {
      // Revert on error
      dispatch({
        type: 'UPDATE_ORDER_STATUS_REVERT',
        payload: { orderId }
      });
    }
  };
  
  return (
    <div>
      <OrderFilters 
        searchTerm={searchTerm}
        onSearchChange={setSearchTerm}
        statusFilter={statusFilter}
        onStatusChange={setStatusFilter}
      />
      <OrderTable 
        orders={filteredOrders}
        onStatusUpdate={handleStatusUpdate}
      />
    </div>
  );
}

  FETCH_[RESOURCE]_REQUEST,
  FETCH_[RESOURCE]_SUCCESS,
  FETCH_[RESOURCE]_FAILURE
} from '../actions/[resource]Actions';

const initialState = {
  [resources]: [],
  loading: false,  // Module-specific loading
  error: null
};

export default function [resource]Reducer(state = initialState, action) {
  switch (action.type) {
    case FETCH_[RESOURCE]_REQUEST:
      return { ...state, loading: true, error: null };
      
    case FETCH_[RESOURCE]_SUCCESS:
      return { ...state, loading: false, [resources]: action.payload, error: null };
      
    case FETCH_[RESOURCE]_FAILURE:
      return { ...state, loading: false, error: action.payload };
      
    default:
      return state;
  }
}

export const fetchProducts = () => async (dispatch) => {
  dispatch({ type: FETCH_PRODUCTS_REQUEST });
  
  try {
    const response = await api.get('/products');
    dispatch({ 
      type: FETCH_PRODUCTS_SUCCESS, 
      payload: response.data 
    });
  } catch (error) {
    dispatch({ 
      type: FETCH_PRODUCTS_FAILURE, 
      payload: error.response?.data?.message || 'Failed to fetch' 
    });
  }
};

Reducer File:
// lib/store/reducers/productReducer.js
import {
  FETCH_PRODUCTS_REQUEST,
  FETCH_PRODUCTS_SUCCESS,
  FETCH_PRODUCTS_FAILURE
} from '../actions/productActions';

const initialState = {
  products: [],
  loading: false,
  error: null
};

export default function productReducer(state = initialState, action) {
  switch (action.type) {
    case FETCH_PRODUCTS_REQUEST:
      return { ...state, loading: true, error: null };
      
    case FETCH_PRODUCTS_SUCCESS:
      return { ...state, loading: false, products: action.payload };
      
    case FETCH_PRODUCTS_FAILURE:
      return { ...state, loading: false, error: action.payload };
      
    default:
      return state;
  }
}

// lib/hooks/useDebounce.js
"use client";  // If it will be used in client components

import { useState, useEffect } from 'react';

export function useDebounce(value, delay = 300) {
  const [debouncedValue, setDebouncedValue] = useState(value);

  useEffect(() => {
    const handler = setTimeout(() => {
      setDebouncedValue(value);
    }, delay);

    return () => clearTimeout(handler);
  }, [value, delay]);

  return debouncedValue;
}

// lib/store/middleware/errorMiddleware.js
const errorMiddleware = (store) => (next) => (action) => {
  if (action.type.endsWith('_FAILURE') || action.error) {
    const errorMessage = action.payload || 'An error occurred';
    console.error('Redux Error:', action.type, errorMessage);
    
    // Dispatch to global error state
    store.dispatch({
      type: 'SET_GLOBAL_ERROR',
      payload: errorMessage
    });
  }
  
  return next(action);
};

export default errorMiddleware;
